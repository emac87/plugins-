# One headless test program per plug-in, linked against that plug-in's shared code.
function(plugin_test name plugin)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE
        $<TARGET_PROPERTY:${plugin},INCLUDE_DIRECTORIES> ${CMAKE_SOURCE_DIR}/${plugin}/Source ${CMAKE_SOURCE_DIR}/Shared)
    target_compile_definitions(${name} PRIVATE $<TARGET_PROPERTY:${plugin},COMPILE_DEFINITIONS>)
    target_link_libraries(${name} PRIVATE ${plugin})
    add_test(NAME ${name} COMMAND ${name} ${CMAKE_BINARY_DIR}/test-output)
endfunction()
enable_testing()
if(BUILD_CONTOUR_EQ)
    plugin_test(contour_tests ContourEQ)
endif()
if(BUILD_CAPTURE_RACK)
    plugin_test(rack_tests CaptureRack)
    target_link_libraries(rack_tests PRIVATE nam_core)
endif()
if(BUILD_OMNISAMPLER)
    plugin_test(omni_tests Omnisampler)
endif()
